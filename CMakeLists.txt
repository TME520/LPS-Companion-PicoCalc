cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico CACHE STRING "Original PicoCalc RP2040 module")
set(PICO_PLATFORM rp2040 CACHE STRING "RP2040 only")
if(NOT PICO_BOARD STREQUAL "pico" OR NOT PICO_PLATFORM STREQUAL "rp2040")
  message(FATAL_ERROR "This project targets the original RP2040 PicoCalc only.")
endif()
set(PICO_SDK_PATH "${CMAKE_CURRENT_LIST_DIR}/deps/pico-sdk" CACHE PATH "Pico SDK")
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(lps_companion C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(PICOTOOL_GIT_BRANCH a7eb3988f0645239185fadb4e25d8279478c2dbb CACHE STRING "Pinned picotool 2.2.0")
pico_sdk_init()
set(CLOCKWORK_PATH "${CMAKE_CURRENT_LIST_DIR}/deps/PicoCalc" CACHE PATH "ClockworkPi source")
set(FATFS_PATH "${CMAKE_CURRENT_LIST_DIR}/deps/FatFs" CACHE PATH "SD library")
set(HELLO "${CLOCKWORK_PATH}/Code/picocalc_helloworld")
add_subdirectory("${FATFS_PATH}/FatFs_SPI" fatfs)
add_executable(lps_companion src/main.cpp src/hw_config.c
  "${HELLO}/lcdspi/lcdspi.c" "${HELLO}/i2ckbd/i2ckbd.c")
target_include_directories(lps_companion PRIVATE "${HELLO}/lcdspi" "${HELLO}/i2ckbd")
target_link_libraries(lps_companion pico_stdlib hardware_spi hardware_i2c FatFs_SPI)
pico_enable_stdio_uart(lps_companion 0)
pico_enable_stdio_usb(lps_companion 0)
# Reserve 64 KiB for the bounded serialization/validation call stack.
# Keep .bss/heap in the first 192 KiB, core0 stack in the next 64 KiB.
file(READ "${PICO_SDK_PATH}/src/rp2_common/pico_crt0/rp2040/memmap_default.ld" MAP)
string(REPLACE "RAM(rwx) : ORIGIN =  0x20000000, LENGTH = 256k" "RAM(rwx) : ORIGIN =  0x20000000, LENGTH = 192k" MAP "${MAP}")
string(REPLACE "SCRATCH_Y(rwx) : ORIGIN = 0x20041000, LENGTH = 4k" "SCRATCH_Y(rwx) : ORIGIN = 0x20030000, LENGTH = 64k" MAP "${MAP}")
if(NOT MAP MATCHES "LENGTH = 192k" OR NOT MAP MATCHES "LENGTH = 64k")
 message(FATAL_ERROR "Unexpected SDK linker layout. Use the pinned SDK.")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/lps_memmap.ld" "${MAP}")
pico_set_linker_script(lps_companion "${CMAKE_CURRENT_BINARY_DIR}/lps_memmap.ld")
target_compile_definitions(lps_companion PRIVATE PICO_STACK_SIZE=0x10000)
pico_set_program_version(lps_companion "1.1")
pico_add_extra_outputs(lps_companion)
